cmake_minimum_required(VERSION 4.0)

project(holomq_main LANGUAGES CXX)

include(CTest)

find_package(CUDAToolkit REQUIRED)
find_package(holoscan REQUIRED CONFIG
    PATHS
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/public/install-cu13-x86_64"
)

add_executable(holomq_main
    main.cpp
    ../common/holoargs.cpp
)
target_compile_features(holomq_main PRIVATE cxx_std_20)
target_include_directories(holomq_main PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}"
    "${CMAKE_CURRENT_LIST_DIR}/../common"
)
target_link_libraries(holomq_main PRIVATE holoscan::holomq CUDA::cuda_driver)

if(BUILD_TESTING)
    find_program(HOLOMQ_BASH_EXECUTABLE bash REQUIRED)
    add_test(
        NAME holomq_main_interprocess_test
        COMMAND
            "${HOLOMQ_BASH_EXECUTABLE}"
            "${CMAKE_CURRENT_LIST_DIR}/test_holomq_main.sh"
            "$<TARGET_FILE:holomq_main>"
    )
    set_tests_properties(holomq_main_interprocess_test PROPERTIES
        LABELS "cuda;holomq;interprocess"
        RUN_SERIAL TRUE
        TIMEOUT 60
    )
endif()
