cmake_minimum_required(VERSION 3.20)
project(operator_status_tracking CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(operator_status_tracking
  operator_status_tracking.cpp
)

target_link_libraries(operator_status_tracking
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_OPERATOR_STATUS_TRACKING_TEST
           COMMAND operator_status_tracking
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_OPERATOR_STATUS_TRACKING_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Application completed. All operators have finished processing."
                       FAIL_REGULAR_EXPRESSION "error"
                      )
endif()
