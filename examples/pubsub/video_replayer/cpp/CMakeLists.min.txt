cmake_minimum_required(VERSION 3.20)
project(pubsub_video_replayer CXX)

find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(pubsub_video_replayer
  pubsub_video_replayer.cpp
)

target_link_libraries(pubsub_video_replayer
  PRIVATE
  holoscan::core
  holoscan::pubsub_runtime
  holoscan::pubsub_fastdds
  holoscan::ops::video_stream_replayer
  holoscan::ops::holoviz
)

# Copy config file
add_custom_target(pubsub_video_replayer_yaml
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/video_replayer.yaml" ${CMAKE_CURRENT_BINARY_DIR}
  DEPENDS "video_replayer.yaml"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/video_replayer.yaml"
)
add_dependencies(pubsub_video_replayer pubsub_video_replayer_yaml)
