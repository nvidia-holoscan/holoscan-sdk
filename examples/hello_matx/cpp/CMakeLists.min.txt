cmake_minimum_required(VERSION 4.0)

if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
    set(CMAKE_CUDA_ARCHITECTURES "native")
endif()

project(holoscan_hello_matx LANGUAGES CXX CUDA)

include(CTest)

find_package(CUDAToolkit REQUIRED)
find_package(holoscan
    PATHS
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/install-${CMAKE_SYSTEM_PROCESSOR}"
    REQUIRED
)

find_package(CCCL 3.3.3 CONFIG REQUIRED)
# Use an installed MatX when CMake finds one (for example through CMAKE_PREFIX_PATH or matx_DIR);
# otherwise download the MatX revision the SDK is tested with (see cmake/deps/matx.cmake).
# FIND_PACKAGE_ARGS enables the find_package(matx CONFIG) attempt; without it, FetchContent's
# default (FETCHCONTENT_TRY_FIND_PACKAGE_MODE=OPT_IN) always downloads.
include(FetchContent)
FetchContent_Declare(matx
    GIT_REPOSITORY https://github.com/NVIDIA/MatX.git
    GIT_TAG 186c72d4794ead29527f45ca4a6f59caa7a08caf
    FIND_PACKAGE_ARGS CONFIG
)
FetchContent_MakeAvailable(matx)

add_executable(hello_matx hello_matx.cu)
target_compile_features(hello_matx PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(hello_matx
    PRIVATE
        CCCL::CCCL
        CUDA::cudart
        holoscan::core
        matx::matx
)

if(BUILD_TESTING)
    add_test(NAME EXAMPLE_CPP_HELLO_MATX_TEST COMMAND hello_matx)
    set_tests_properties(EXAMPLE_CPP_HELLO_MATX_TEST
        PROPERTIES
            PASS_REGULAR_EXPRESSION "MatX CUDA sample complete"
            SKIP_RETURN_CODE 77
    )
endif()
