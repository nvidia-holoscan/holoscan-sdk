cmake_minimum_required(VERSION 4.0)

project(holoscan_hello_schema LANGUAGES CXX)

include(CTest)

find_package(holoscan REQUIRED CONFIG
    PATHS
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/install-${CMAKE_SYSTEM_PROCESSOR}"
)

add_executable(hello_schema hello_schema.cpp)
target_compile_features(hello_schema PRIVATE cxx_std_20)
# holoscan::sensor_io supplies the Image schema and its metadata helper; it is a separate package
# component from holoscan-core, so an install that omitted it will not configure here.
target_link_libraries(hello_schema
    PRIVATE
        holoscan::schema
        holoscan::sensor_io
)

if(BUILD_TESTING)
    add_test(NAME EXAMPLE_CPP_HELLO_SCHEMA_TEST COMMAND hello_schema)
    set_tests_properties(EXAMPLE_CPP_HELLO_SCHEMA_TEST
        PROPERTIES
            PASS_REGULAR_EXPRESSION "holoscan schema sample complete"
    )
endif()
