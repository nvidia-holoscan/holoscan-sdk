cmake_minimum_required(VERSION 3.20)
project(ping_period CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_periodic
  ping_periodic.cpp
)

target_link_libraries(ping_periodic
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_PING_PERIODIC_TEST
           COMMAND ping_periodic
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_PING_PERIODIC_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Rx message received: Periodic ping")
endif()
