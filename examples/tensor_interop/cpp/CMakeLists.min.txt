cmake_minimum_required(VERSION 3.20)
project(holoscan_tensor_interop CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(tensor_interop
  tensor_interop.cpp
  receive_tensor_gxf.hpp
  send_tensor_gxf.hpp
)

target_link_libraries(tensor_interop
  PRIVATE
  holoscan::core
  holoscan::data_loggers::basic_console_logger
  CUDA::cudart
)

# Testing
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME EXAMPLE_CPP_TENSOR_INTEROP_TEST
    COMMAND bash -c
      "${CMAKE_CURRENT_BINARY_DIR}/tensor_interop 2>&1 | \
      ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../bin/test_pattern_validation.py --strip-quotes \
        --pass-pattern 'Scheduling 3 entities' \
        --pass-pattern 'BasicConsoleLogger\\[ID:mx.in\\]' \
        --pass-pattern 'BasicConsoleLogger\\[ID:mx.out\\]' \
        --pass-pattern 'Category:TensorMap' \
        --pass-pattern '\\[Stream: none\\]' \
        --pass-pattern 'TensorMap\\(size=1\\) \\{tensor: Tensor\\(shape=\\[4, 4, 3\\], dtype=kDLUInt:8:1, device=kDLCUDA:0, size=48, nbytes=48, C-contiguous=true\\)\\}' \
        --fail-pattern 'error'"
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  )
endif()
