cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_any CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_any
  ping_any.cpp
)

target_link_libraries(ping_any
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_PING_ANY_TEST
           COMMAND ping_any
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_PING_ANY_TEST PROPERTIES
                      PASS_REGULAR_EXPRESSION "Rx message value: 30")
endif()
