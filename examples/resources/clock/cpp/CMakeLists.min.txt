cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_clock CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_clock
  ping_clock.cpp
)

target_link_libraries(ping_clock
  PRIVATE
  holoscan::core
  holoscan::ops::ping_rx
  holoscan::ops::ping_tx
)

# Testing
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)

  add_test(NAME EXAMPLE_RESOURCES_CPP_PING_CLOCK_TEST
    COMMAND bash -c
      "${CMAKE_CURRENT_BINARY_DIR}/ping_clock 2>&1 | \
      ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../bin/test_pattern_validation.py \
        --pass-pattern 'Rx message value: 3' \
        --pass-pattern 'final timestamp'"
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  )
endif()
