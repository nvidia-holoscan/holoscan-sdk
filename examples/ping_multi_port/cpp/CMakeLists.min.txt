cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_multi_port CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_multi_port
  ping_multi_port.cpp
)

target_link_libraries(ping_multi_port
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_PING_MULTI_PORT_TEST
           COMMAND ping_multi_port
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_PING_MULTI_PORT_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Rx message value1: 57")
endif()
