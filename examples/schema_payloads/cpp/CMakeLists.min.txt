cmake_minimum_required(VERSION 4.0)

project(holoscan_schema_payloads LANGUAGES CXX)

include(CTest)

find_package(Threads REQUIRED)
find_package(holoscan
    PATHS
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/install-${CMAKE_SYSTEM_PROCESSOR}"
    REQUIRED
)

add_executable(schema_payloads schema_payloads.cpp)
target_compile_features(schema_payloads PRIVATE cxx_std_20)
# holoscan::sensor_io supplies the Imu schema this example carries; it is a separate package
# component from holoscan-core, so an install that omitted it will not configure here.
target_link_libraries(schema_payloads
    PRIVATE holoscan::core holoscan::schema holoscan::sensor_io Threads::Threads)

if(BUILD_TESTING)
    # REQ: SYSTEM-014
    add_test(NAME EXAMPLE_CPP_SCHEMA_PAYLOADS_TEST COMMAND schema_payloads)
    set_tests_properties(EXAMPLE_CPP_SCHEMA_PAYLOADS_TEST
        PROPERTIES
            PASS_REGULAR_EXPRESSION
                "schema_payloads: strings=5 times=4 imus=4 degraded=1 oversize=kResourceExhausted termination=stopped"
            LABELS "examples;schema"
            TIMEOUT 30)
endif()
