cmake_minimum_required(VERSION 3.20)
project(pose_tree_basic CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(pose_tree_basic
  pose_tree_basic.cpp
)

target_link_libraries(pose_tree_basic
  PRIVATE
  holoscan::core
  holoscan::pose_tree
)

# Copy config file
add_custom_target(pose_tree_basic_yaml
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/pose_tree_basic.yaml" ${CMAKE_CURRENT_BINARY_DIR}
  DEPENDS "pose_tree_basic.yaml"
  BYPRODUCTS "pose_tree_basic.yaml"
)
add_dependencies(pose_tree_basic pose_tree_basic_yaml)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_POSE_TREE_BASIC
           COMMAND pose_tree_basic
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_POSE_TREE_BASIC PROPERTIES
                       PASS_REGULAR_EXPRESSION "day 365"
                       FAIL_REGULAR_EXPRESSION "error"
                      )
endif()
