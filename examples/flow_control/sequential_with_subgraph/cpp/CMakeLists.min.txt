cmake_minimum_required(VERSION 3.20)
project(sequential_with_subgraph CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(sequential_with_subgraph
  sequential_with_subgraph.cpp
)

target_link_libraries(sequential_with_subgraph
  PRIVATE
  holoscan::core
)

# Testing

           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_SEQUENTIAL_WITH_SUBGRAPH_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "I am here - node4"
                       FAIL_REGULAR_EXPRESSION "error"
                      )
endif()
