cmake_minimum_required(VERSION 3.20)
project(mixed_flows CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(mixed_flows_execution
  mixed_flows_execution.cpp
)

target_link_libraries(mixed_flows_execution
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_MIXED_FLOWS_EXECUTION_TEST
           COMMAND mixed_flows_execution
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_MIXED_FLOWS_EXECUTION_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Sending value 1 - node1.*(I am here - alarm.*Received value 1 - node2|Received value 1 - node2.*I am here - alarm).*Sending value 2 - node1.*(I am here - alarm.*Received value 2 - node3|Received value 2 - node3.*I am here - alarm)")
endif()
