cmake_minimum_required(VERSION 3.20)
project(fork_join CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(fork_join_execution
  fork_join_execution.cpp
)

target_link_libraries(fork_join_execution
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_FORK_JOIN_EXECUTION_TEST
           COMMAND fork_join_execution
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_FORK_JOIN_EXECUTION_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "executing operator: node1.*executing operator: node[2-6].*executing operator: node7.*executing operator: node8")
endif()
