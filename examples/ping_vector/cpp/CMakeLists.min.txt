cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_vector CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_vector
  ping_vector.cpp
)

target_link_libraries(ping_vector
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_PING_VECTOR_TEST
           COMMAND ping_vector
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_PING_VECTOR_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Rx message receiver value\\[1\\]\\[4\\]: 126")
endif()
