cmake_minimum_required(VERSION 3.20)
project(holoscan_gpu_resident_multi_io LANGUAGES CXX CUDA)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(gpu_resident_multi_io
  gpu_resident_multi_io.cpp
  multi_io_kernels.cu
)

target_link_libraries(gpu_resident_multi_io
  PRIVATE
  holoscan::core
)

# Set CUDA architectures for the target
set_property(TARGET gpu_resident_multi_io PROPERTY CUDA_ARCHITECTURES "native")

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_GPU_RESIDENT_MULTI_IO_TEST
           COMMAND gpu_resident_multi_io
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_GPU_RESIDENT_MULTI_IO_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "GPU-resident multi-IO example completed")
endif()
