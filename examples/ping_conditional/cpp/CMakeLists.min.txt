cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_conditional CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_conditional
  ping_conditional.cpp
)

target_link_libraries(ping_conditional
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)

  add_test(NAME EXAMPLE_CPP_PING_CONDITIONAL_TEST
    COMMAND bash -c
      "${CMAKE_CURRENT_BINARY_DIR}/ping_conditional 2>&1 | \
      ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../bin/test_pattern_validation.py \
        --pass-pattern 'Rx message value: 27' \
        --pass-pattern 'Rx did not receive any value'"
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  )
endif()
