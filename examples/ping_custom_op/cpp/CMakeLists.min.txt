cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_custom_op CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_custom_op
  ping_custom_op.cpp
)

target_link_libraries(ping_custom_op
  PRIVATE
  holoscan::core
  holoscan::ops::ping_rx
  holoscan::ops::ping_tx
)

# Testing
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)

  add_test(NAME EXAMPLE_CPP_PING_CUSTOM_OP_TEST
    COMMAND bash -c
      "${CMAKE_CURRENT_BINARY_DIR}/ping_custom_op 2>&1 | \
      ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/../bin/test_pattern_validation.py \
        --pass-pattern 'Middle message value: 10' \
        --pass-pattern 'Rx message value: 30'"
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  )
endif()
