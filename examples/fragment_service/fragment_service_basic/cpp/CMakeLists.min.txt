cmake_minimum_required(VERSION 3.20)
project(fragment_service_basic CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(fragment_service_basic
  fragment_service_basic.cpp
)

target_link_libraries(fragment_service_basic
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_FRAGMENT_SERVICE_BASIC_TEST
           COMMAND fragment_service_basic
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_FRAGMENT_SERVICE_BASIC_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "MyService value: 10"
                       FAIL_REGULAR_EXPRESSION "error"
                      )
endif()
