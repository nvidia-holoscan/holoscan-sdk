cmake_minimum_required(VERSION 4.0)

project(holoscan_contract_on_aligned LANGUAGES CXX)

include(CTest)

find_package(holoscan
    PATHS
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/install-${CMAKE_SYSTEM_PROCESSOR}"
    REQUIRED
)

add_executable(contract_on_aligned contract_on_aligned.cpp)
target_link_libraries(contract_on_aligned PRIVATE holoscan::core)

if(BUILD_TESTING)
    # REQ: SYSTEM-013
    add_test(NAME EXAMPLE_CPP_CONTRACT_ON_ALIGNED_TEST COMMAND contract_on_aligned)
    set_tests_properties(EXAMPLE_CPP_CONTRACT_ON_ALIGNED_TEST
        PROPERTIES
            PASS_REGULAR_EXPRESSION "fired=4 skipped_left=4 skipped_right=4"
            LABELS "examples;temporal-contract"
            TIMEOUT 30)
endif()
