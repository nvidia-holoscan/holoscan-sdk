cmake_minimum_required(VERSION 3.20)
project(holoscan_ping_periodic_async_buffer CXX)

# Finds the package holoscan
find_package(holoscan REQUIRED CONFIG
             PATHS "/opt/nvidia/holoscan" "/workspace/holoscan-sdk/install")

add_executable(ping_periodic_async_buffer
  ping_periodic_async_buffer.cpp
)

target_link_libraries(ping_periodic_async_buffer
  PRIVATE
  holoscan::core
)

# Testing
if(BUILD_TESTING)
  add_test(NAME EXAMPLE_CPP_PING_PERIODIC_ASYNC_BUFFER_TEST
           COMMAND ping_periodic_async_buffer
           WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
          )
  set_tests_properties(EXAMPLE_CPP_PING_PERIODIC_ASYNC_BUFFER_TEST PROPERTIES
                       PASS_REGULAR_EXPRESSION "Rx message received: 10")
endif()
