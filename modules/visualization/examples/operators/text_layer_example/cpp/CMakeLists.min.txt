cmake_minimum_required(VERSION 4.0)

if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
    set(CMAKE_CUDA_ARCHITECTURES "native")
endif()

project(holoscan_text_layer_example LANGUAGES CXX CUDA)

find_package(Threads REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(holoscan REQUIRED CONFIG COMPONENTS visualization
    PATHS
        "/opt/nvidia/holoscan"
        "/workspace/holoscan-sdk/install"
        "/workspace/holoscan-sdk/public/install-cu13-x86_64"
)

if(NOT TARGET holoscan::visualization::op_holoviz)
    message(FATAL_ERROR
        "text_layer_example needs holoscan::visualization::op_holoviz; install the SDK with the "
        "visualization module enabled (HOLOSCAN_BUILD_VISUALIZATION=ON) including the "
        "holoscan-visualization component.")
endif()

add_executable(text_layer_example
    text_layer_example.cpp
    text_source.cu
)
target_compile_features(text_layer_example PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(text_layer_example
    PRIVATE
        CUDA::cudart
        holoscan::core
        holoscan::sensor_io
        holoscan::visualization::op_holoviz
        Threads::Threads
)
