cmake_minimum_required(VERSION 3.30.4)
project(holoscan_descriptor_schema_diagnostic LANGUAGES CXX)

foreach(required_variable IN ITEMS
        DIAGNOSTIC_CASE
        HOLOSCAN_FLATC_EXECUTABLE
        HOLOSCAN_FLATC_VERSION
        HOLOSCAN_SCHEMA_DIR
        HOLOSCAN_REFLECTION_SCHEMA
        HOLOSCAN_SCHEMA_TOOLS_DIR)
    if(NOT DEFINED ${required_variable} OR "${${required_variable}}" STREQUAL "")
        message(FATAL_ERROR "${required_variable} is required")
    endif()
endforeach()

add_executable(flatbuffers::flatc IMPORTED GLOBAL)
set_target_properties(flatbuffers::flatc PROPERTIES
    IMPORTED_LOCATION "${HOLOSCAN_FLATC_EXECUTABLE}"
)
add_library(holoscan::fbs_tools INTERFACE IMPORTED GLOBAL)
set_target_properties(holoscan::fbs_tools PROPERTIES
    HOLOSCAN_FLATC_VERSION "${HOLOSCAN_FLATC_VERSION}"
    HOLOSCAN_SCHEMA_DIR "${HOLOSCAN_SCHEMA_DIR}"
    HOLOSCAN_FLATBUFFERS_REFLECTION_SCHEMA "${HOLOSCAN_REFLECTION_SCHEMA}"
)
include("${HOLOSCAN_SCHEMA_TOOLS_DIR}/HoloscanSchemaTools.cmake")

if(DIAGNOSTIC_CASE STREQUAL "missing_tensor")
    set(diagnostic_schema missing_tensor.fbs)
    set(diagnostic_root app3p::MissingTensor)
elseif(DIAGNOSTIC_CASE STREQUAL "object_vector")
    set(diagnostic_schema object_vector.fbs)
    set(diagnostic_root app3p::ObjectVector)
elseif(DIAGNOSTIC_CASE STREQUAL "wrong_native")
    set(diagnostic_schema valid_descriptor.fbs)
    set(diagnostic_root app3p::ValidDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "union")
    set(diagnostic_schema union.fbs)
    set(diagnostic_root app3p::UnionDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "recursive_table")
    set(diagnostic_schema recursive_table.fbs)
    set(diagnostic_root app3p::RecursiveDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "nested_tensor")
    set(diagnostic_schema nested_tensor.fbs)
    set(diagnostic_root app3p::NestedTensorDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "required_string")
    set(diagnostic_schema required_string.fbs)
    set(diagnostic_root app3p::RequiredStringDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "required_vector")
    set(diagnostic_schema required_vector.fbs)
    set(diagnostic_root app3p::RequiredVectorDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "required_table")
    set(diagnostic_schema required_table.fbs)
    set(diagnostic_root app3p::RequiredTableDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "required_tensor")
    set(diagnostic_schema required_tensor.fbs)
    set(diagnostic_root app3p::RequiredTensorDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "force_align")
    set(diagnostic_schema force_align.fbs)
    set(diagnostic_root app3p::ForceAlignDescriptor)
elseif(DIAGNOSTIC_CASE STREQUAL "file_identifier")
    set(diagnostic_schema file_identifier.fbs)
    set(diagnostic_root app3p::IdentifiedDescriptor)
else()
    message(FATAL_ERROR "Unknown DIAGNOSTIC_CASE: ${DIAGNOSTIC_CASE}")
endif()

if(DIAGNOSTIC_CASE STREQUAL "wrong_native")
    set(diagnostic_native app3p::NotTheGeneratedNativeT)
else()
    set(diagnostic_native "${diagnostic_root}T")
endif()
string(REGEX REPLACE "^.*::" "" diagnostic_root_basename "${diagnostic_root}")

holoscan_add_flatbuffer_schema(
    TARGET diagnostic_schema_generated
    SCHEMAS "${diagnostic_schema}"
    INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated"
    BFBS_EMBED
    DESCRIPTOR_TENSOR
    TRAITS_HEADER diagnostic_traits.hpp
    TRAITS_ROOT_TYPE "${diagnostic_root}"
    TRAITS_NATIVE_TYPE "${diagnostic_native}"
    TRAITS_BINARY_SCHEMA_TYPE "app3p::${diagnostic_root_basename}BinarySchema"
    TRAITS_PACKAGE_NAMESPACE app3p::diagnostic_package
    TRAITS_CANONICAL_NAME "app3p.${diagnostic_root_basename}"
    TRAITS_SCHEMA_VERSION 1
    TRAITS_MAX_SERIALIZED_SIZE 512
)
